import requests

KUMAR_API = "https://kumar-lac.vercel.app/api/bot"
USER_ID = ""

def get_points():
    payload = {"userId": USER_ID, "action": "getScore"}
    try:
        response = requests.post(KUMAR_API, json=payload)
        return response.json().get('score', 0)
    except:
        return "Error"
